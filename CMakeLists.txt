cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(VirtualCOMWorkbench VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/SDK" CACHE PATH "Vdevice SDK directory; override privately for production")
option(ENABLE_ASAN "AddressSanitizer for offline validation" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /PDBALTPATH:%_PDB%)
if(ENABLE_ASAN)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(CdcCore STATIC App/Cdc.cpp App/SelfTest.cpp "${SDK_DIRECTORY}/Vdevice.cpp")
target_include_directories(CdcCore PUBLIC App "${SDK_DIRECTORY}")
add_library(SerialSession STATIC App/Session.cpp)
target_link_libraries(SerialSession PUBLIC CdcCore setupapi cfgmgr32 advapi32 ole32)
foreach(target VirtualCOMWorkbench VirtualCOMWorkbenchCLI)
 if(target STREQUAL "VirtualCOMWorkbench")
  add_executable(${target} WIN32 App/Main.cpp)
 else()
  add_executable(${target} App/Main.cpp)
  target_compile_definitions(${target} PRIVATE CONSOLE_ENTRY)
 endif()
 target_link_libraries(${target} PRIVATE SerialSession comctl32 shell32 dwmapi gdiplus)
 target_link_options(${target} PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/App/App.manifest")
endforeach()
enable_testing()
add_executable(CdcTests Tests/ModelTests.cpp)
target_link_libraries(CdcTests PRIVATE CdcCore)
add_test(NAME CdcModelAndByteRouting COMMAND CdcTests)
add_test(NAME OfflineApplication COMMAND VirtualCOMWorkbenchCLI --self-test)
